{{- /* Given a course root, return its lessons in reading order with computed numbers (1.a, 1.b, …). Call with partialCached and the course RelPermalink as the variant key. */ -}} {{- $letters := split "abcdefghijklmnopqrstuvwxyz" "" -}} {{- $lessons := slice -}} {{- $previous := false -}} {{- range $module := .Sections.ByWeight -}} {{- if gt (len $module.RegularPages) (len $letters) -}} {{- errorf "%s: module %q has %d lessons; lesson letters run out after %d (a … z)" $module.File.Path $module.LinkTitle (len $module.RegularPages) (len $letters) -}} {{- end -}} {{- range $i, $lesson := $module.RegularPages.ByWeight -}} {{- if $lesson.Date.IsZero -}} {{- errorf "%s: a lesson needs a publication date" $lesson.File.Path -}} {{- end -}} {{- if $previous -}} {{- if not ($lesson.Date.After $previous.Date) -}} {{- errorf "%s: publication date must be after %s in reading order" $lesson.File.Path $previous.File.Path -}} {{- end -}} {{- end -}} {{- if $lesson.Lastmod.Before $lesson.Date -}} {{- errorf "%s: updated date cannot precede publication" $lesson.File.Path -}} {{- end -}} {{- $lessons = $lessons | append (dict "page" $lesson "module" $module "number" (printf "%d.%s" $module.Weight (index $letters $i))) -}} {{- $previous = $lesson -}} {{- end -}} {{- end -}} {{- return $lessons -}}
